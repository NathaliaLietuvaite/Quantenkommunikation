                                  navigator_rcf: float,
                                  bridge_data: Optional[Dict[str, Any]] = None

        """
        # D* ranges from ~0.001 (very coherent) to ~0.05 (needs more diffusion)
        return max(0.001, (1.0 - navigator_rcf) * 0.05)

    def integrate(self,
                  navigator_state: np.ndarray,
                  navigator_rcf: float,
                  extracted_features: np.ndarray,
                  little_vector: LittleVector) -> Tuple[np.ndarray, float, float, float]:
        """

        Parameters:

        - navigator_rcf: Current RCF of the Navigator

        features = extracted_features
        if features.shape != navigator_state.shape:
            # Pad or truncate to match

            features = features[:min_dim]
            nav = navigator_state[:min_dim]
        else:
            nav = navigator_state

        denom = navigator_rcf + D_star

            padded = np.zeros_like(navigator_state)
            padded[:len(new_state)] = new_state
            new_state = padded
            new_state = new_state / np.linalg.norm(new_state)

        new_rcf = little_vector.compute_rcf(new_state)
        logger.info(f"  New RCF after superposition: {new_rcf:.4f} "
